function doGet(e) {
  try {
    const p = (e && e.parameter) || {};
    const action = String(p.action || 'health').trim();
    if (action === 'health') return out_({ok:true, service:'KaenatChi Booking Core', mode:'staging-rebuild', cmsWrites:false});
    if (action === 'getConfig') return out_(getPublicConfig_());
    if (action === 'getSlots') return out_(getSlots_(p.date, String(p.serviceIds||'').split(',').filter(Boolean)));
    if (action === 'getStatus') return out_(getStatus_(p.trackingCode, p.phone, p.requestId));
    return out_({ok:false,error:'UNKNOWN_ACTION',message:'عملیات شناخته نشد.'});
  } catch (err) {
    console.error(err);
    return out_({ok:false,error:'SERVER_ERROR',message:'خطای داخلی سرور.'});
  }
}


/**
 * Read-only CMS connection/contract check for STAGING.
 * Makes one GET request to the published CMS API through getPublicConfig_().
 * Does not write to Sheets, Script Properties, CMS, VIP, or Telegram.
 */
function auditStagingCmsReadOnly() {
  const report = {
    ok: false,
    readOnly: true,
    cmsReachable: false,
    contractValid: false,
    serviceCount: 0,
    servicesWithStableIds: 0,
    servicesWithValidPrices: 0,
    error: ''
  };

  try {
    const config = getPublicConfig_();
    const services = Array.isArray(config.services) ? config.services : [];
    report.cmsReachable = true;
    report.contractValid = true;
    report.serviceCount = services.length;
    report.servicesWithStableIds = services.filter(s =>
      s && typeof s.id === 'string' && s.id.trim()
    ).length;
    report.servicesWithValidPrices = services.filter(s =>
      s && Number.isFinite(Number(s.price)) && Number(s.price) >= 0
    ).length;
    report.ok = services.length > 0 &&
      report.servicesWithStableIds === services.length &&
      report.servicesWithValidPrices === services.length;
    if (!report.ok) report.error = 'SERVICE_DATA_VALIDATION_FAILED';
  } catch (err) {
    report.error = String(err && (err.code || err.message) || 'CMS_UNAVAILABLE');
  }

  console.log(JSON.stringify(report, null, 2));
  return report;
}

function doPost(e) {
  try {
    const raw = e && e.postData && e.postData.contents;
    const body = raw ? JSON.parse(raw) : {};
    if (body && body.callback_query) return out_(handleTelegramCallback_(body.callback_query, e && e.parameter));
    const action = String(body.action || '').trim();
    if (action === 'submitQuestionnaire') return out_(submitQuestionnaire_(body));
    if (action === 'getQuestionnaireStatus') return out_(getQuestionnaireStatus_(body));
    if (action === 'getBookingGrant') return out_(getBookingGrant_(body));
    if (action === 'reviewQuestionnaire') { requireAdmin_(body.adminSecret); return out_(reviewQuestionnaire_(body)); }
    if (action === 'revokeBookingGrant') { requireAdmin_(body.adminSecret); return out_(revokeBookingGrant_(body)); }
    if (action === 'createRequest') return out_(createRequest_(body));
    if (action === 'submitReceipt') return out_(submitReceipt_(body));
    if (action === 'adminDecision') {
      requireAdmin_(body.adminSecret);
      return out_(adminDecision_(body));
    }
    if (action === 'requestCancel') return out_(requestCancel_(body));
    if (action === 'requestReschedule') return out_(requestReschedule_(body));
    if (action === 'managerRescheduleOffer') { requireAdmin_(body.adminSecret); return out_(managerRescheduleOffer_(body)); }
    if (action === 'acceptManagerReschedule') return out_(acceptManagerReschedule_(body));
    return out_({ok:false,error:'UNKNOWN_ACTION',message:'عملیات شناخته نشد.'});
  } catch (err) {
    console.error(err);
    return out_({ok:false,error:err.code || 'SERVER_ERROR',message:err.publicMessage || 'درخواست انجام نشد.'});
  }
}

/** Run manually only in the Apps Script project intended for STAGING. */
function initializeStagingSchema() {
  const ss = staging_();
  Object.keys(CORE.HEADERS).forEach(name => {
    let sh = ss.getSheetByName(name);
    if (!sh) sh = ss.insertSheet(name);
    const expected = CORE.HEADERS[name];
    const lastColumn = sh.getLastColumn();
    const current = sh.getLastRow() && lastColumn
      ? sh.getRange(1,1,1,lastColumn).getDisplayValues()[0].map(v => String(v || '').trim())
      : [];
    const blank = current.every(v => !v);
    if (blank) {
      sh.getRange(1,1,1,expected.length).setValues([expected]);
    } else {
      // Staging-only additive migration: preserve existing columns and rows.
      const missing = expected.filter(h => current.indexOf(h) < 0);
      if (missing.length) sh.getRange(1,lastColumn+1,1,missing.length).setValues([missing]);
    }
  });
  return {ok:true, spreadsheet:ss.getName(), tabs:Object.keys(CORE.HEADERS), note:'No CMS tabs or values were changed.'};
}


function getPublicConfig_() {
  // Read published CMS data only. Never writes to CMS.
  const api = prop_(CORE.CMS_API_URL_PROP);

  if (!api) {
    fail_(
      'CMS_UNAVAILABLE',
      'نشانی API کائنات‌چی در تنظیمات خصوصی Staging پیکربندی نشده است.'
    );
  }

  let response;

  try {
    response = UrlFetchApp.fetch(
      api + (api.indexOf('?') >= 0 ? '&' : '?') +
        'action=getMiniAppData',
      {
        muteHttpExceptions: true,
        followRedirects: true
      }
    );
  } catch (err) {
    fail_('CMS_UNAVAILABLE', 'ارتباط با API کائنات‌چی برقرار نشد.');
  }

  if (
    response.getResponseCode() < 200 ||
    response.getResponseCode() >= 300
  ) {
    fail_('CMS_UNAVAILABLE', 'API کائنات‌چی پاسخ معتبر نداد.');
  }

  let payload;

  try {
    payload = JSON.parse(response.getContentText());
  } catch (err) {
    fail_('CMS_UNAVAILABLE', 'پاسخ API کائنات‌چی JSON معتبر نیست.');
  }

  if (
    !payload ||
    payload.success !== true ||
    !Array.isArray(payload.services)
  ) {
    fail_(
      'CMS_CONTRACT_MISMATCH',
      'ساختار دادهٔ خدمات CMS با قرارداد Staging هماهنگ نیست.'
    );
  }

  const mapped = payload.services
    .map(function (r, index) {
      const name = String(
        first_(r, [
          'نام خدمت', 'عنوان', 'نام',
          'Service Name', 'Name', 'Title', 'title'
        ]) || ''
      ).trim();

      const explicitId = String(
        first_(r, [
          'شناسه', 'شناسه خدمت', 'Service ID',
          'ServiceId', 'ID', 'Id', 'id', 'slug'
        ]) || ''
      ).trim();

      const rawPrice = first_(r, [
        'قیمت', 'هزینه', 'Price',
        'Base Price', 'Original Price', 'مبلغ', 'price'
      ]);

      const active = String(
        first_(r, [
          'فعال', 'وضعیت', 'Active',
          'active', 'Status', 'status'
        ]) || ''
      ).trim().toLocaleLowerCase('fa');

      const normalizedName = name.replace(
        /[\u200c\u200f\u200e\s_-]+/g,
        ''
      ).toLocaleLowerCase('fa');

      const isActive =
        !active ||
        ['بله', 'فعال', 'true', 'yes', '1', 'active']
          .includes(active);

      const isAggregate = [
        'انرژیخوانی',
        'شمعتراپی',
        'سایکوتراپی'
      ].includes(normalizedName);

      const hasPrice =
        rawPrice !== undefined &&
        rawPrice !== null &&
        String(rawPrice).trim() !== '';

      const price = num_(rawPrice);

      // Stable fallback independent of CMS row order. Duplicate IDs fail closed below.
      const id = explicitId || bookingSlug_(name);

      return {
        id: id,
        name: name,
        category: String(
          first_(r, [
            'دسته', 'دسته‌بندی',
            'Category', 'Service Category', 'category'
          ]) || ''
        ),
        description: String(
          first_(r, [
            'توضیح کوتاه', 'Description',
            'Short Description', 'توضیحات', 'description'
          ]) || ''
        ),
        price: price,
        duration:
          num_(first_(r, [
            'مدت', 'مدت زمان', 'Duration', 'duration'
          ])) || 30,
        _active: isActive,
        _hasPrice: hasPrice,
        _aggregate: isAggregate
      };
    })
    .filter(function (s) {
      return (
        s.id &&
        s.name &&
        s._active &&
        s._hasPrice &&
        !s._aggregate
      );
    })
    .map(function (s) {
      return {
        id: s.id,
        // Keep both names during the staging contract transition:
        // the backend's canonical field is "name", while some clients consume "title".
        name: s.name,
        title: s.name,
        category: s.category,
        description: s.description,
        price: s.price,
        duration: s.duration
      };
    });

  const serviceIds = mapped.map(function (s) { return s.id; });
  if (new Set(serviceIds).size !== serviceIds.length) fail_('CMS_SERVICE_ID_DUPLICATE','شناسهٔ خدمات در CMS یکتا نیست؛ رزرو تا اصلاح شناسه‌ها غیرفعال است.');

  if (!mapped.length) {
    fail_(
      'CMS_SERVICES_EMPTY',
      'API کائنات‌چی هیچ خدمت قابل رزروی برنگرداند.'
    );
  }

  return {
    ok: true,
    services: mapped,
    availableDates: [tomorrowJalali_()],
    bookingRules: {
      requestReviewHours: 12,
      paymentMinutes: 30,
      timezone: CORE.TZ
    }
  };
}

/** Slot generation follows the approved weekly schedule; CMS closures are overlaid read-only. */
function getSlots_(date, serviceIds, excludeRequestId) {
  const tomorrow = tomorrowJalali_();
  if (String(date || tomorrow) !== tomorrow) return {ok:true,date:tomorrow,slots:[],message:'فقط برای فردا امکان ثبت درخواست وجود دارد.'};
  const day=tehranTomorrowWeekday_();
  const ids=Array.isArray(serviceIds)?[...new Set(serviceIds.map(String))]:[];
  let schedule=scheduleForDay_(day),duration=30,serviceMode='energy';
  if(ids.length) {
    const catalog=getPublicConfig_().services;
    const selected=ids.map(id=>catalog.find(s=>s.id===id));
    if(selected.some(s=>!s)) return {ok:true,date:tomorrow,slots:[],message:'یکی از خدمات انتخاب‌شده معتبر نیست.'};
    const isTherapy=s=>/درمان|therapy|psychotherap|روان.?درمان|شمع.?درمان/i.test((s.category||'')+' '+(s.name||''));
    const therapy=selected.some(isTherapy),energy=selected.some(s=>!isTherapy(s));
    if(therapy&&energy) return {ok:true,date:tomorrow,slots:[],message:'خدمات درمانی و انرژی در یک نوبت قابل ترکیب نیستند.'};
    serviceMode=therapy?'therapy':'energy';
    duration=selected.reduce((sum,s)=>sum+Math.max(1,num_(s.duration)||30),0);
    if(therapy) schedule=day===4?therapySlots_():[];
  }
  const closures=readClosureSet_();
  const slots=schedule.map(time=>{
    const key=tomorrow+'|'+time;
    const closed=closures.dates.has(tomorrow)||closures.slots.has(key);
    const withinHours=slotWithinServiceWindow_(day,time,duration);
    const taken=!withinHours||slotHasConflict_(tomorrow,time,duration,excludeRequestId||'');
    const status=closed?'BLOCKED':!withinHours?'BLOCKED':taken?'BOOKED':'AVAILABLE';
    return {date:tomorrow,time,period:(time<'15:00'?'صبح':'عصر'),status,
      label:closed?'تعطیل':!withinHours?'مدت خدمت خارج از ساعت کاری':taken?'رزرو شده':time,
      message:closed?'این زمان در این تاریخ قابل رزرو نیست.':!withinHours?'مدت مجموع خدمات از ساعت کاری این بازه فراتر می‌رود.':taken?'این زمان با نوبت دیگری تداخل دارد؛ زمان دیگری انتخاب کن.':''};
  });
  return {ok:true,date:tomorrow,slots,serviceMode,durationMinutes:duration};
}

/**
 * Questionnaire/grant workflow. User identity is accepted only from signed Telegram WebApp initData.
 * All data is isolated to the staging spreadsheet; no CMS or VIP writes are performed.
 */
const QUESTIONNAIRE_SERVICES_ = Object.freeze({
  'candle-therapy': {name:'شمع‌تراپی', topics:[
    'مسائل عاطفی و رابطه‌ای','جدایی، دلتنگی و وابستگی','ترس‌ها و نگرانی‌های شخصی',
    'اعتمادبه‌نفس و ارزشمندی خود','احساس گیر افتادن یا تکرار الگوها',
    'هدف‌ها، تصمیم‌ها و موانع شخصی','موضوعات دیگر با توضیح آزاد'
  ]},
  'psychotherapy': {name:'سایکوتراپی', topics:[]}
});
const QUESTIONNAIRE_STATUS_ = Object.freeze({
  SUBMITTED:'SUBMITTED', UNDER_REVIEW:'UNDER_REVIEW',
  CONSULTATION_PENDING:'CONSULTATION_PENDING', APPROVED:'APPROVED', DECLINED:'DECLINED'
});
const GRANT_STATE_ = Object.freeze({ACTIVE:'ACTIVE',CONSUMED:'CONSUMED',REVOKED:'REVOKED'});

function submitQuestionnaire_(b) {
  return withLock_(() => {
    const identity=verifyTelegramInitData_(String(b.telegramInitData||''));
    if(!identity) fail_('TELEGRAM_IDENTITY_REQUIRED','برای ارسال فرم، مینی‌اپ را از داخل تلگرام باز کن.');
    const serviceKey=String(b.serviceKey||'').trim();
    const service=QUESTIONNAIRE_SERVICES_[serviceKey];
    if(!service) fail_('INVALID_QUESTIONNAIRE_SERVICE','خدمت انتخاب‌شده برای این فرم معتبر نیست.');
    const submissionId=String(b.submissionId||'').trim();
    if(!/^QST-[A-Z0-9-]{8,64}$/i.test(submissionId)) fail_('INVALID_SUBMISSION_ID','شناسه فرم معتبر نیست.');
    const topic=String(b.topic||'').trim();
    if(serviceKey==='candle-therapy'&&!service.topics.includes(topic)) fail_('INVALID_TOPIC','یک موضوع معتبر انتخاب کن.');
    const answers=b.answers&&typeof b.answers==='object'&&!Array.isArray(b.answers)?b.answers:{};
    const safeAnswers={};
    const allowed=serviceKey==='candle-therapy'
      ? ['issue','duration','dailyImpact','pastActions','desiredChange','additional']
      : ['topic','lifeImpact','pastActions','desiredOutcome','additional'];
    allowed.forEach(k=>{
      const value=String(answers[k]===undefined?'':answers[k]).trim();
      if(value.length>3000) fail_('ANSWER_TOO_LONG','هر پاسخ باید حداکثر ۳۰۰۰ نویسه باشد.');
      if(value) safeAnswers[k]=value;
    });
    const required=serviceKey==='candle-therapy'
      ? ['issue','duration','dailyImpact','pastActions','desiredChange']
      : ['topic','lifeImpact','pastActions','desiredOutcome'];
    if(required.some(k=>!safeAnswers[k])) fail_('QUESTIONNAIRE_INCOMPLETE','لطفاً همه پرسش‌های اصلی فرم را تکمیل کن.');
    const sh=tab_(CORE.TABS.QUESTIONNAIRES);
    const existing=objectsWithRows_(sh).find(x=>String(x.obj['Submission ID'])===submissionId);
    if(existing) {
      if(String(existing.obj['Telegram User ID'])!==identity.id || String(existing.obj['Service Key'])!==serviceKey)
        fail_('SUBMISSION_ID_CONFLICT','این شناسه قبلاً برای فرم دیگری ثبت شده است.');
      return questionnairePublicResult_(existing.obj,true);
    }
    const now=new Date();
    const row={
      'Submission ID':submissionId,'Telegram User ID':identity.id,'Telegram Username':identity.username||'',
      'Service Key':serviceKey,'Service Name':service.name,'Topic':topic,'Answers JSON':JSON.stringify(safeAnswers),
      'Status':QUESTIONNAIRE_STATUS_.SUBMITTED,'Submitted At':now,'Reviewed By':'','Reviewed At':'','Decision Note':''
    };
    appendObject_(sh,row);
    questionnaireAudit_(submissionId,'','SUBMITTED','telegram-user',{serviceKey,answerKeys:Object.keys(safeAnswers)});
    notifyQuestionnaireAdmin_(row);
    return questionnairePublicResult_(row,false);
  });
}

function getQuestionnaireStatus_(b) {
  const identity=verifyTelegramInitData_(String(b.telegramInitData||''));
  if(!identity) fail_('TELEGRAM_IDENTITY_REQUIRED','هویت تلگرام تأیید نشد.');
  const id=String(b.submissionId||'').trim();
  const row=objectsWithRows_(tab_(CORE.TABS.QUESTIONNAIRES)).find(x=>String(x.obj['Submission ID'])===id);
  if(!row||String(row.obj['Telegram User ID'])!==identity.id) fail_('NOT_FOUND','فرم پیدا نشد.');
  return questionnairePublicResult_(row.obj,true);
}
function questionnairePublicResult_(row,recovered) {
  return {ok:true,submissionId:String(row['Submission ID']),serviceKey:String(row['Service Key']),
    serviceName:String(row['Service Name']),status:String(row.Status),
    submittedAt:row['Submitted At']?new Date(row['Submitted At']).toISOString():null,
    recovered:!!recovered,message:'فرم دریافت شد؛ دسترسی به رزرو فقط پس از تأیید دستی مایا فعال می‌شود.'};
}

function reviewQuestionnaire_(b) {
  return withLock_(() => {
    const id=String(b.submissionId||'').trim();
    const decision=String(b.decision||'').trim();
    const allowed=['consultation','approve','decline'];
    if(!id||!allowed.includes(decision)) fail_('INVALID_QUESTIONNAIRE_DECISION','تصمیم فرم معتبر نیست.');
    const ref=objectsWithRows_(tab_(CORE.TABS.QUESTIONNAIRES)).find(x=>String(x.obj['Submission ID'])===id);
    if(!ref) fail_('NOT_FOUND','فرم پیدا نشد.');
    const row=ref.obj;
    if(row.Status===QUESTIONNAIRE_STATUS_.APPROVED) {
      if(decision==='approve') return {ok:true,status:row.Status,grant:getGrantRow_(String(row['Telegram User ID']),String(row['Service Key']))?.obj||null,message:'فرم قبلاً تأیید شده است.'};
      fail_('INVALID_STATE','فرم تأییدشده را نمی‌توان با این تصمیم تغییر داد.');
    }
    if(row.Status===QUESTIONNAIRE_STATUS_.DECLINED) {
      if(decision==='decline') return {ok:true,status:row.Status,message:'این فرم قبلاً رد شده است.'};
      fail_('INVALID_STATE','فرم ردشده باید با ارسال فرم تازه بررسی شود.');
    }
    const now=new Date(),actor='admin';
    if(decision==='consultation') {
      row.Status=QUESTIONNAIRE_STATUS_.CONSULTATION_PENDING;
      row['Decision Note']=String(b.note||'').trim().slice(0,500);
      row['Reviewed By']=actor;row['Reviewed At']=now;
      writeObject_(ref.sheet,ref.row,row);
      questionnaireAudit_(id,'','CONSULTATION_PENDING',actor,{});
      notifyQuestionnaireCustomer_(row,'نیاز به گفت‌وگو','مایا فرم را دیده و برای گفت‌وگوی اولیه با تو هماهنگ می‌کند. هنوز مجوز رزرو صادر نشده است.');
      return {ok:true,status:row.Status,message:'برای گفت‌وگو علامت‌گذاری شد؛ مجوز رزرو صادر نشد.'};
    }
    if(decision==='decline') {
      row.Status=QUESTIONNAIRE_STATUS_.DECLINED;row['Decision Note']=String(b.note||'').trim().slice(0,500);
      row['Reviewed By']=actor;row['Reviewed At']=now;writeObject_(ref.sheet,ref.row,row);
      questionnaireAudit_(id,'','DECLINED',actor,{});
      notifyQuestionnaireCustomer_(row,'نتیجه بررسی فرم','در حال حاضر دسترسی رزرو برای این خدمت تأیید نشد. اگر نیاز باشد، مایا درباره ادامه مسیر با تو هماهنگ می‌کند.');
      return {ok:true,status:row.Status,message:'فرم رد شد؛ مجوزی ساخته نشد.'};
    }
    const priorGrant=getGrantRow_(String(row['Telegram User ID']),String(row['Service Key']));
    if(priorGrant&&String(priorGrant.obj.State)==='ACTIVE'&&String(priorGrant.obj['Submission ID'])!==id)
      fail_('ACTIVE_GRANT_EXISTS','برای همین خدمت و کاربر از قبل مجوز فعال وجود دارد.');
    let grant=priorGrant&&String(priorGrant.obj['Submission ID'])===id?priorGrant.obj:null;
    if(!grant) {
      grant={'Grant ID':'GRT-'+Utilities.getUuid().replace(/-/g,'').slice(0,24).toUpperCase(),
        'Telegram User ID':String(row['Telegram User ID']),'Service Key':String(row['Service Key']),
        'Submission ID':id,'State':GRANT_STATE_.ACTIVE,'Granted By':actor,'Granted At':now,
        'Consumed At':'','Consumed Request ID':'','Revoked At':'','Revocation Reason':''};
      // Write the grant first. A retry can find this grant by submission ID if the next write fails.
      appendObject_(tab_(CORE.TABS.GRANTS),grant);
    }
    row.Status=QUESTIONNAIRE_STATUS_.APPROVED;row['Decision Note']=String(b.note||'').trim().slice(0,500);
    row['Reviewed By']=actor;row['Reviewed At']=now;writeObject_(ref.sheet,ref.row,row);
    questionnaireAudit_(id,String(grant['Grant ID']),'APPROVED_GRANT_ISSUED',actor,{serviceKey:row['Service Key']});
    notifyQuestionnaireCustomer_(row,'فرم تأیید شد','مایا فرم را تأیید کرده است. مجوز فقط برای همین خدمت صادر شده؛ این به معنی قطعی‌شدن نوبت یا تأیید پرداخت نیست.');
    return {ok:true,status:row.Status,grantState:grant.State,message:'فرم تأیید شد و مجوز مخصوص همین خدمت صادر شد.'};
  });
}

function handleQuestionnaireCallback_(q,params) {
  const supplied=String((params||{}).telegram_webhook||''),expected=prop_(CORE.WEBHOOK_SECRET_PROP);
  if(!expected||supplied!==expected) return {ok:false,error:'UNAUTHORIZED'};
  const adminChat=prop_(CORE.ADMIN_CHAT_PROP);
  const sender=String(q&&q.from&&q.from.id||''),chat=String(q&&q.message&&q.message.chat&&q.message.chat.id||'');
  if(!adminChat||sender!==adminChat||chat!==adminChat) return {ok:false,error:'UNAUTHORIZED',message:'فقط مدیر مجاز به بررسی فرم است.'};
  const parts=String(q.data||'').split(':');
  if(parts.length!==3||parts[0]!=='qreview') return {ok:false,error:'INVALID_CALLBACK'};
  try {
    const result=reviewQuestionnaire_({submissionId:parts[2],decision:parts[1]});
    telegram_('answerCallbackQuery',{callback_query_id:q.id,text:'ثبت شد'});
    return result;
  } catch(err) {
    telegram_('answerCallbackQuery',{callback_query_id:q.id,text:err.publicMessage||'انجام نشد'});
    throw err;
  }
}

function createRequest_(b) {
  return withLock_(() => {
    const firstName = String(b.firstName || '').trim();
    const lastName = String(b.lastName || '').trim();
    const phone = String(b.phone || '');
    if (!/^[\u0621-\u063A\u0641-\u064A\u067E\u0686\u0698\u06A9\u06AF\u06CC\u200c\s]+$/.test(firstName) || !/^[\u0621-\u063A\u0641-\u064A\u067E\u0686\u0698\u06A9\u06AF\u06CC\u200c\s]+$/.test(lastName)) fail_('INVALID_NAME','نام و نام خانوادگی را فقط با حروف فارسی وارد کن.');
    if (!/^09[0-9]{9}$/.test(phone)) fail_('INVALID_PHONE','شماره همراه باید دقیقاً ۱۱ رقم انگلیسی و با 09 شروع شود.');
    const date = String(b.date || '');
    const time = normalizeTime_(b.time);
    if (date !== tomorrowJalali_()) fail_('DATE_NOT_ALLOWED','فقط برای فردا می‌توان درخواست ثبت کرد.');
    const ids = Array.isArray(b.serviceIds) ? [...new Set(b.serviceIds.map(String))] : [];
    if (!ids.length) fail_('SERVICE_REQUIRED','حداقل یک خدمت انتخاب کن.');
    const catalog = getPublicConfig_().services;
    const chosen = ids.map(id => catalog.find(s => s.id === id));
    if (chosen.some(s => !s)) fail_('INVALID_SERVICE','یکی از خدمات انتخاب‌شده معتبر نیست.');
    const subtotal = chosen.reduce((sum,s) => sum + s.price,0);
    const telegramIdentity = verifyTelegramInitData_(String(b.telegramInitData || ''));
    const discount = validateDiscountReadOnly_(String(b.discountCode || ''), subtotal);
    const suppliedRequestId = String(b.requestId || '').trim();
    if (suppliedRequestId && !/^REQ-[A-Z0-9-]{8,64}$/i.test(suppliedRequestId)) fail_('INVALID_REQUEST_ID','شناسهٔ درخواست معتبر نیست.');
    const requestId = suppliedRequestId || ('REQ-' + Utilities.getUuid().replace(/-/g,'').slice(0,20).toUpperCase());
    const prior = findByRequestId_(requestId);
    if (prior) {
      const same = String(prior.obj.Phone)===phone && String(prior.obj['First Name'])===firstName && String(prior.obj['Last Name'])===lastName && String(prior.obj['Appointment Date'])===date && normalizeTime_(prior.obj['Appointment Time'])===time && String(prior.obj['Service IDs']).split(',').filter(Boolean).sort().join(',')===ids.slice().sort().join(',') && String(prior.obj['Discount Code']||'').trim().toUpperCase()===String(b.discountCode||'').trim().toUpperCase();
      if (!same) fail_('REQUEST_ID_CONFLICT','این شناسه قبلاً برای درخواست دیگری استفاده شده است.');
      return {ok:true,requestId:requestId,trackingCode:String(prior.obj['Tracking Code']||''),status:String(prior.obj.Status||''),expiresAt:prior.obj['Request Expires At'] ? new Date(prior.obj['Request Expires At']).toISOString() : null,total:Number(prior.obj.Total||0),discountAmount:Number(prior.obj['Discount Amount']||0),message:'درخواست قبلی بازیابی شد؛ درخواست تکراری ساخته نشد.'};
    }
    const slot = getSlots_(date, ids).slots.find(s => s.time === time);
    if (!slot || slot.status !== 'AVAILABLE') fail_('SLOT_UNAVAILABLE','این زمان دیگر در دسترس نیست؛ زمان دیگری انتخاب کن.');
    const now = new Date();
    const tracking = Utilities.getUuid().replace(/-/g,'').slice(0,24).toUpperCase();
    const expires = new Date(now.getTime()+CORE.REQUEST_TTL_MS);
    const row = {
      'Request ID':requestId,'Tracking Code':tracking,'Appointment Code':'','Status':CORE.STATUS.REQUEST_PENDING,
      'First Name':firstName,'Last Name':lastName,'Phone':phone,'Appointment Date':date,'Appointment Time':time,
      'Service IDs':ids.join(','),'Service Names':chosen.map(s=>s.name).join('، '),'Subtotal':subtotal,
      'Discount Code':discount.code,'Discount Amount':discount.amount,'Total':Math.max(0,subtotal-discount.amount),
      'Created At':now,'Request Expires At':expires,'Payment Expires At':'','Receipt File ID':'','Receipt MIME Type':'',
      'Transaction Ref':'','Decision At':'','Decision By':'','Reschedule From':'',
      'Telegram Chat ID':telegramIdentity ? String(telegramIdentity.id) : '',
      'Telegram Username':telegramIdentity ? String(telegramIdentity.username || '') : '',
      'Updated At':now
    };
    appendObject_(tab_(CORE.TABS.BOOKINGS),row);
    log_(requestId,'REQUEST_CREATED','',CORE.STATUS.REQUEST_PENDING,'customer',{date,time,services:ids,total:row.Total});
    notifyAdmin_(requestId,'درخواست نوبت جدید',row);
    return {ok:true,requestId,trackingCode:tracking,status:CORE.STATUS.REQUEST_PENDING,expiresAt:expires.toISOString(),total:row.Total,discountAmount:discount.amount,message:'درخواست ثبت شد؛ نوبت هنوز قطعی نیست و تا پذیرش، پرداختی لازم نیست.'};
  });
}

function submitReceipt_(b) {
  return withLock_(() => {
    const rowRef = findBooking_(String(b.trackingCode||''),String(b.phone||''));
    if (!rowRef) fail_('NOT_FOUND','درخواست پیدا نشد؛ کد پیگیری و شماره همراه را بررسی کن.');
    const row = rowRef.obj;
    if (row.Status !== CORE.STATUS.PAYMENT_PENDING) fail_('INVALID_STATE','این درخواست در مرحله دریافت فیش نیست.');
    if (new Date(row['Payment Expires At']).getTime() < Date.now()) {
      transition_(rowRef,CORE.STATUS.EXPIRED,'system','PAYMENT_WINDOW_EXPIRED');
      fail_('PAYMENT_EXPIRED','مهلت ۳۰ دقیقه‌ای پرداخت تمام شده است؛ اگر زمان آزاد باشد، درخواست تازه ثبت کن.');
    }
    const mime = String(b.mimeType||'');
    const base64 = String(b.receiptBase64||'');
    const tx = String(b.transactionRef||'').trim();
    if (!/^image\/(jpeg|png|webp)$/.test(mime) || !base64 || base64.length > 7000000) fail_('INVALID_RECEIPT','تصویر فیش معتبر لازم است.');
    if (!tx) fail_('TRANSACTION_REF_REQUIRED','شماره پیگیری تراکنش را وارد کن.');
    const bytes = Utilities.base64Decode(base64);
    const blob = Utilities.newBlob(bytes,mime,'receipt-'+row['Request ID']);
    const file = DriveApp.createFile(blob);
    file.setSharing(DriveApp.Access.PRIVATE,DriveApp.Permission.VIEW);
    row['Receipt File ID'] = file.getId(); row['Receipt MIME Type'] = mime; row['Transaction Ref'] = tx; row.Status = CORE.STATUS.RECEIPT_REVIEW; row['Updated At'] = new Date();
    writeObject_(rowRef.sheet,rowRef.row,row);
    appendObject_(tab_(CORE.TABS.PAYMENTS),{'Payment ID':'PAY-'+Utilities.getUuid(),'Request ID':row['Request ID'],'Tracking Code':row['Tracking Code'],'Amount':row.Total,'Transaction Ref':tx,'Receipt File ID':file.getId(),'Receipt MIME Type':mime,'Status':'RECEIPT_REVIEW','Submitted At':new Date(),'Reviewed At':'','Reviewed By':'','Notes':''});
    log_(row['Request ID'],'RECEIPT_SUBMITTED',CORE.STATUS.PAYMENT_PENDING,CORE.STATUS.RECEIPT_REVIEW,'customer',{transactionRef:tx});
    notifyReceipt_(row,file.getBlob());
    return {ok:true,status:CORE.STATUS.RECEIPT_REVIEW,message:'فیش دریافت شد؛ نوبت تا بررسی واقعی پرداخت توسط مایا نگه داشته می‌شود.'};
  });
}

function adminDecision_(b) {
  return withLock_(() => {
    const id = String(b.requestId||'');
    const ref = findByRequestId_(id);
    if (!ref) fail_('NOT_FOUND','درخواست پیدا نشد.');
    const row = ref.obj;
    const decision = String(b.decision||'');
    const actor = String(b.actor||'maya');
    if (decision === 'accept') {
      if (row.Status !== CORE.STATUS.REQUEST_PENDING) fail_('INVALID_STATE','این درخواست دیگر در انتظار پذیرش اولیه نیست.');
      if (new Date(row['Request Expires At']).getTime() < Date.now()) {
        transition_(ref,CORE.STATUS.EXPIRED,'system','REQUEST_WINDOW_EXPIRED');
        fail_('REQUEST_EXPIRED','مهلت بررسی درخواست تمام شده است.');
      }
      row['Payment Expires At'] = new Date(Date.now()+CORE.PAYMENT_TTL_MS);
      transition_(ref,CORE.STATUS.PAYMENT_PENDING,actor,'ADMIN_ACCEPTED');
      row['Updated At'] = new Date(); writeObject_(ref.sheet,ref.row,row);
      notifyCustomerPaymentWindow_(row);
      return {ok:true,status:row.Status,paymentDeadline:row['Payment Expires At'].toISOString()};
    }
    if (decision === 'reject') {
      if (![CORE.STATUS.REQUEST_PENDING,CORE.STATUS.RECEIPT_REVIEW].includes(row.Status)) fail_('INVALID_STATE','رد در این مرحله مجاز نیست.');
      transition_(ref,CORE.STATUS.REJECTED,actor,'ADMIN_REJECTED');
      row['Decision At']=new Date();row['Decision By']=actor;row['Updated At']=new Date();writeObject_(ref.sheet,ref.row,row);
      notifyCustomer_(row,'درخواست نوبت پذیرفته نشد','این درخواست به تأیید نرسید. برای هماهنگی بیشتر می‌توانی با مایا در ارتباط باشی.\nکد پیگیری: '+row['Tracking Code']);
      return {ok:true,status:row.Status};
    }
    if (decision === 'approvePayment') {
      if (row.Status !== CORE.STATUS.RECEIPT_REVIEW) fail_('INVALID_STATE','این فیش در انتظار بررسی نیست.');
      // Never confirm a discounted booking until VIP token consumption is
      // implemented atomically with payment approval. Otherwise a token could
      // remain reusable after the booking is confirmed.
      if (String(row['Discount Code']||'').trim()) {
        fail_('DISCOUNT_FINALIZATION_UNAVAILABLE','تأیید نهایی این پرداخت تا تکمیل مصرف امن توکن VIP متوقف است؛ نوبت همچنان برای بررسی مدیر نگه داشته می‌شود.');
      }
      row['Appointment Code'] = 'KC-' + Utilities.getUuid().replace(/-/g,'').slice(0,10).toUpperCase();
      row['Decision At']=new Date();row['Decision By']=actor;row['Updated At']=new Date();
      transition_(ref,CORE.STATUS.CONFIRMED,actor,'BANK_DEPOSIT_VERIFIED');
      writeObject_(ref.sheet,ref.row,row);
      updatePaymentStatus_(id,'APPROVED',actor);
      notifyCustomer_(row,'پرداختت تأیید شد؛ نوبت قطعی است 🎉','واریز بررسی و تأیید شد.\nکد نوبت: '+row['Appointment Code']+'\nزمان: '+row['Appointment Date']+' '+row['Appointment Time']+'\nکد پیگیری: '+row['Tracking Code']);
      return {ok:true,status:row.Status,appointmentCode:row['Appointment Code']};
    }
    if (decision === 'rejectPayment') {
      if (row.Status !== CORE.STATUS.RECEIPT_REVIEW) fail_('INVALID_STATE','این فیش در انتظار بررسی نیست.');
      transition_(ref,CORE.STATUS.REJECTED,actor,'PAYMENT_REJECTED');
      row['Decision At']=new Date();row['Decision By']=actor;row['Updated At']=new Date();writeObject_(ref.sheet,ref.row,row);
      updatePaymentStatus_(id,'REJECTED',actor);
      notifyCustomer_(row,'فیش پرداخت نیاز به پیگیری دارد','فیش ارسالی تأیید نشد. لطفاً برای بررسی و هماهنگی با مایا در ارتباط باش.\nکد پیگیری: '+row['Tracking Code']);
      return {ok:true,status:row.Status};
    }
    if (decision === 'approveReschedule') {
      if (row.Status !== CORE.STATUS.RESCHEDULE_REQUESTED) fail_('INVALID_STATE','درخواست تغییر زمان در صف بررسی نیست.');
      const d=String(row['Requested Reschedule Date']||'');const t=normalizeTime_(row['Requested Reschedule Time']);
      if(!isSlotAvailableFor_(d,t,String(row['Service IDs']||'').split(',').filter(Boolean),id)) fail_('SLOT_UNAVAILABLE','زمان پیشنهادی دیگر در دسترس نیست؛ نوبت قبلی حفظ شده است.');
      row['Appointment Date']=d;row['Appointment Time']=t;row['Appointment Code']='KC-'+Utilities.getUuid().replace(/-/g,'').slice(0,10).toUpperCase();row['Requested Reschedule Date']='';row['Requested Reschedule Time']='';row.Status=CORE.STATUS.CONFIRMED;row['Updated At']=new Date();writeObject_(ref.sheet,ref.row,row);
      log_(id,'RESCHEDULE_APPROVED',CORE.STATUS.RESCHEDULE_REQUESTED,CORE.STATUS.CONFIRMED,actor,{newDate:d,newTime:t,oldSlotReleased:true});
      notifyCustomer_(row,'تغییر زمان نوبت تأیید شد','زمان جدید نوبتت ثبت شد.\nکد نوبت: '+row['Appointment Code']+'\nزمان جدید: '+d+' '+t+'\nکد پیگیری: '+row['Tracking Code']);
      return {ok:true,status:row.Status,date:d,time:t,appointmentCode:row['Appointment Code']};
    }
    if (decision === 'rejectReschedule') {
      if (row.Status !== CORE.STATUS.RESCHEDULE_REQUESTED) fail_('INVALID_STATE','درخواست تغییر زمان در صف بررسی نیست.');
      row['Requested Reschedule Date']='';row['Requested Reschedule Time']='';row.Status=CORE.STATUS.CONFIRMED;row['Updated At']=new Date();writeObject_(ref.sheet,ref.row,row);
      log_(id,'RESCHEDULE_REJECTED',CORE.STATUS.RESCHEDULE_REQUESTED,CORE.STATUS.CONFIRMED,actor,{oldSlotRetained:true});
      notifyCustomer_(row,'درخواست تغییر زمان تأیید نشد','نوبت قبلی‌ات همچنان برقرار است. برای هماهنگی بیشتر با مایا در ارتباط باش.\nکد پیگیری: '+row['Tracking Code']);
      return {ok:true,status:row.Status,message:'درخواست تغییر زمان رد شد؛ نوبت قبلی حفظ شد.'};
    }
    if (decision === 'approveCancel') {
      if (row.Status !== CORE.STATUS.CANCEL_REQUESTED) fail_('INVALID_STATE','درخواست لغو در صف بررسی نیست.');
      transition_(ref,CORE.STATUS.CANCELLED,actor,'CANCELLATION_APPROVED');
      row['Decision At']=new Date();row['Decision By']=actor;row['Updated At']=new Date();writeObject_(ref.sheet,ref.row,row);
      notifyCustomer_(row,'درخواست لغو ثبت شد','لغو نوبت بررسی شد. اگر بازپرداخت لازم باشد، انتقال وجه به‌صورت دستی انجام می‌شود. برای هماهنگی با مایا در ارتباط باش.\nکد پیگیری: '+row['Tracking Code']);
      return {ok:true,status:row.Status,refundRequiresManualProcessing:true,message:'لغو ثبت شد؛ بازپرداخت کارت‌به‌کارت باید دستی انجام شود.'};
    }
    if (decision === 'rejectCancel') {
      if (row.Status !== CORE.STATUS.CANCEL_REQUESTED) fail_('INVALID_STATE','درخواست لغو در صف بررسی نیست.');
      row.Status=CORE.STATUS.CONFIRMED;row['Decision At']=new Date();row['Decision By']=actor;row['Updated At']=new Date();writeObject_(ref.sheet,ref.row,row);
      log_(id,'CANCELLATION_REJECTED',CORE.STATUS.CANCEL_REQUESTED,CORE.STATUS.CONFIRMED,actor,{});
      notifyCustomer_(row,'نوبتت همچنان برقرار است','درخواست لغو تأیید نشد و نوبت قبلی حفظ شده است. برای هماهنگی با مایا در ارتباط باش.\nکد پیگیری: '+row['Tracking Code']);
      return {ok:true,status:row.Status,message:'درخواست لغو رد شد؛ نوبت حفظ شد.'};
    }
    if (decision === 'holdForReview') {
      if (row.Status !== CORE.STATUS.RECEIPT_REVIEW) fail_('INVALID_STATE','فیشی برای نگه‌داشتن در صف بررسی وجود ندارد.');
      log_(id,'PAYMENT_HELD_FOR_REVIEW',row.Status,row.Status,actor,{});
      return {ok:true,status:row.Status,autoRelease:false};
    }
    fail_('INVALID_DECISION','تصمیم مدیریتی نامعتبر است.');
  });
}

function requestCancel_(b) {
  return withLock_(() => {
    const ref=findBooking_(String(b.trackingCode||''),String(b.phone||''));
    if(!ref) fail_('NOT_FOUND','درخواست پیدا نشد.');
    if(ref.obj.Status!==CORE.STATUS.CONFIRMED) fail_('INVALID_STATE','درخواست لغو فقط برای نوبت قطعی قابل ثبت است.');
    // Customer cancellation is always a request for Maya to review; no automatic

function requestReschedule_(b) {
  return withLock_(() => {
    const ref=findBooking_(String(b.trackingCode||''),String(b.phone||''));

    if(!rescheduleWindowOpen_(currentAppointmentAt,Date.now())) {
      fail_('RESCHEDULE_WINDOW_CLOSED','درخواست تغییر زمان باید حداقل ۱۲ ساعت پیش از نوبت ثبت شود.');

    ref.obj['Requested Reschedule Time']=targetTime;

    ref.obj['Updated At']=new Date();
    writeObject_(ref.sheet,ref.row,ref.obj);
    log_(ref.obj['Request ID'],'RESCHEDULE_REQUESTED',CORE.STATUS.CONFIRMED,CORE.STATUS.RESCHEDULE_REQUESTED,'customer',{targetDate,targetTime,oldSlotRetained:true});
    notifyAdmin_(ref.obj['Request ID'],'درخواست تغییر زمان',ref.obj);
    return {ok:true,status:CORE.STATUS.RESCHEDULE_REQUESTED,message:'درخواست تغییر زمان بدون جریمه برای بررسی مایا ثبت شد؛ پرداخت مجدد لازم نیست.'};

function validateDiscountReadOnly_(code,subtotal) {
  if(!code) return {code:'',amount:0};
  // Fail closed until the request carries server-verified Telegram initData
  // and token ownership is bound to that verified identity. A client-supplied
  // telegramId is not proof of ownership and must never authorize a VIP discount.
  fail_('DISCOUNT_IDENTITY_UNVERIFIED','اعمال کد تخفیف موقتاً غیرفعال است تا هویت تلگرام و مالکیت توکن به‌صورت امن تأیید شود.');
  const clean=String(code).trim().toUpperCase();
  const vipId=prop_(CORE.VIP_ID_PROP);
  if(!vipId) fail_('DISCOUNT_UNAVAILABLE','اتصال VIP در تنظیمات خصوصی Staging پیکربندی نشده است.');
  const vip=SpreadsheetApp.openById(vipId);
  const ts=vip.getSheetByName('توکن ها');
  const cs=vip.getSheetByName('مشتریان');
  if(!ts||!cs) fail_('DISCOUNT_UNAVAILABLE','امکان بررسی کد تخفیف در حال حاضر وجود ندارد.');
  const tokens=ts.getLastRow()>1?ts.getRange(2,1,ts.getLastRow()-1,Math.max(10,ts.getLastColumn())).getValues():[];
  const customers=cs.getLastRow()>1?cs.getRange(2,1,cs.getLastRow()-1,8).getValues():[];
  const token=tokens.find(r=>String(r[0]||'').trim().toUpperCase()===clean);
  if(!token) fail_('INVALID_DISCOUNT','کد تخفیف معتبر نیست.');
  const customer=customers.find(r=>String(r[0]||'').trim()===String(token[1]||'').trim());
  const status=String(token[5]||'').trim();
  const expiry=Number(token[9]||0);
  if(status==='مصرف‌شده') fail_('DISCOUNT_USED','کد تخفیف قبلاً استفاده شده است.');
  if(status!=='فعال') fail_('DISCOUNT_INACTIVE','کد تخفیف فعال نیست.');
  if(expiry&&expiry<=Date.now()) fail_('DISCOUNT_EXPIRED','کد تخفیف منقضی شده است.');
  if(!customer||String(customer[7]||'').trim()!=='فعال') fail_('DISCOUNT_MEMBER_INACTIVE','عضویت مربوط به این کد تخفیف فعال نیست.');
  const percent=Math.max(0,Math.min(100,num_(token[2])));
  return {code:clean,percent,amount:Math.round(subtotal*percent/100)};
}

function slotWithinServiceWindow_(day,time,duration) {
  const start=minutes_(time);
  if(start<0||!Number.isFinite(Number(duration))||Number(duration)<=0)return false;
  const end=start+Number(duration);
  // Sunday and Tuesday use half-hour start times and their final listed starts
  // (13:30 and 18:30) require the corresponding 14:00 / 19:00 session end.
  const halfHourDay=day===0||day===2;
  const morningEnd=halfHourDay?14*60:13*60+30;
  const eveningEnd=halfHourDay?19*60:18*60+30;
  const inMorning=start>=11*60&&end<=morningEnd;
  const inEvening=start>=16*60&&end<=eveningEnd;
  return inMorning||inEvening;
}
function scheduleForDay_(day) {
  // JS weekday: 0 Sun, 1 Mon, ... 6 Sat
  if(day===5) return [];
  if(day===4) return ['11:00','12:00','13:00','16:00','17:00','18:00'];
  if([6,1,3].includes(day)) return ['11:00','12:00','13:00','16:00','17:00','18:00'];
  if([0,2].includes(day)) return ['11:30','12:30','13:30','16:30','17:30','18:30'];
  return [];
}
function readClosureSet_() {
  // Booking closures are operational data in the isolated booking spreadsheet.
  // This function is read-only and never opens or writes to the CMS spreadsheet.
  const dates=new Set(),slots=new Set();
  const ss=staging_(),sh=ss.getSheetByName(CORE.TABS.CLOSURES);
  if(!sh) fail_('CLOSURES_TAB_MISSING','شیت Closures در Staging موجود نیست؛ تا اصلاح ساختار، رزرو متوقف شد.');
  const rows=readObjects_(ss,[CORE.TABS.CLOSURES]);
  const target=tomorrowJalali_();
  rows.forEach(r=>{
    if(!active_(first_(r,['active','Active','فعال','Status']))) return;
    const rawStart=first_(r,['start_datetime','Start Datetime','start','Start']);
    const rawEnd=first_(r,['end_datetime','End Datetime','end','End']);
    const startDate=closureDate_(rawStart),endDate=closureDate_(rawEnd);
    if(!startDate||!endDate) fail_('CLOSURE_DATE_INVALID','یک بازهٔ مسدود در Staging تاریخ معتبر ندارد؛ برای جلوگیری از رزرو ناامن، بررسی متوقف شد.');
    if(target<startDate||target>endDate) return;
    const startTime=closureTime_(rawStart),endTime=closureTime_(rawEnd);
    if(startDate!==endDate||!startTime||!endTime||startTime>=endTime) {
      dates.add(target);
      return;
    }
    let cursor=minutes_(startTime),finish=minutes_(endTime);
    while(cursor<finish) {
      slots.add(target+'|'+('0'+Math.floor(cursor/60)).slice(-2)+':'+('0'+(cursor%60)).slice(-2));
      cursor+=30;
    }
  });
  return {dates,slots};
}
function closureDate_(value) {
  if(value instanceof Date && Number.isFinite(value.getTime())) {
    const parts=new Intl.DateTimeFormat('en-u-ca-persian',{timeZone:CORE.TZ,year:'numeric',month:'2-digit',day:'2-digit'}).formatToParts(value);
    const p={};parts.forEach(x=>p[x.type]=x.value);return p.year+'/'+p.month+'/'+p.day;
  }
  const s=String(value||'').trim();
  const match=s.match(/\d{4}[\/-]\d{1,2}[\/-]\d{1,2}/);
  const normalized=normalizeDate_(match?match[0]:'');
  if(/^\d{4}\/\d{2}\/\d{2}$/.test(normalized)) return normalized;
  const d=new Date(s);
  if(!Number.isFinite(d.getTime())) return '';
  const parts=new Intl.DateTimeFormat('en-u-ca-persian',{timeZone:CORE.TZ,year:'numeric',month:'2-digit',day:'2-digit'}).formatToParts(d);
  const p={};parts.forEach(x=>p[x.type]=x.value);return p.year+'/'+p.month+'/'+p.day;
}
function closureTime_(value) {
  if(value instanceof Date && Number.isFinite(value.getTime())) {
    return new Intl.DateTimeFormat('en-GB',{timeZone:CORE.TZ,hour:'2-digit',minute:'2-digit',hourCycle:'h23'}).format(value);
  }
  const s=String(value||'').trim();
  const m=s.match(/(?:T|\s)(\d{1,2}:\d{2})(?::\d{2})?/);
  return normalizeTime_(m?m[1]:s);
}
function rescheduleWindowOpen_(appointmentAt,now) {
  const target=Number(appointmentAt),current=Number(now);
  return Number.isFinite(target)&&Number.isFinite(current)&&target-current>=CORE.RESCHEDULE_MIN_NOTICE_MS;
}
function expirationReason_(status,requestExpiry,paymentExpiry,now) {
  const current=Number(now);
  const requestAt=Number(requestExpiry),paymentAt=Number(paymentExpiry);
  if(status===CORE.STATUS.REQUEST_PENDING && Number.isFinite(requestAt) && requestAt>0 && requestAt<=current) return 'REQUEST_WINDOW_EXPIRED';
  if(status===CORE.STATUS.PAYMENT_PENDING && Number.isFinite(paymentAt) && paymentAt>0 && paymentAt<=current) return 'PAYMENT_WINDOW_EXPIRED';
  // RECEIPT_REVIEW is intentionally excluded: a submitted receipt must remain
  // held until a manager finishes bank verification, regardless of elapsed time.
  return '';
}

function isSlotAvailableFor_(date,time,serviceIds,excludeRequestId) {
  if(date!==tomorrowJalali_()) return false;
  const day=tehranTomorrowWeekday_(),ids=Array.isArray(serviceIds)?serviceIds:[];
  if(!ids.length) return false;
  const catalog=getPublicConfig_().services;
  const selected=ids.map(id=>catalog.find(s=>s.id===id));
  // Reject unknown service IDs rather than silently treating them as a shorter booking.
  if(selected.some(s=>!s)) return false;
  const isTherapy=s=>/درمان|therapy|psychotherap|روان.?درمان|شمع.?درمان/i.test((s.category||'')+' '+(s.name||''));
  const therapy=selected.some(isTherapy);
  const energy=selected.some(s=>!isTherapy(s));
  if(therapy&&energy) return false;
  const duration=selected.reduce((sum,s)=>sum+Math.max(1,num_(s.duration)||30),0);
  if(therapy ? day!==4 || !therapySlots_().includes(time) : !scheduleForDay_(day).includes(time)) return false;
  if(!slotWithinServiceWindow_(day,time,duration)) return false;
  const closures=readClosureSet_();
  if(closures.dates.has(date)||closures.slots.has(date+'|'+time)) return false;
  return !slotHasConflict_(date,time,duration,excludeRequestId);
}
function slotHasConflict_(date,time,duration,excludeRequestId) {
  const start=minutes_(time);
  if(start<0)return true;
  const end=start+Math.max(1,Number(duration)||30);
  const active=[CORE.STATUS.REQUEST_PENDING,CORE.STATUS.PAYMENT_PENDING,CORE.STATUS.RECEIPT_REVIEW,CORE.STATUS.CONFIRMED,CORE.STATUS.CANCEL_REQUESTED,CORE.STATUS.RESCHEDULE_REQUESTED,CORE.STATUS.MANAGER_RESCHEDULE_OFFERED];
  const rows=objectsWithRows_(tab_(CORE.TABS.BOOKINGS)).filter(x=>active.includes(String(x.obj.Status||''))&&String(x.obj['Request ID']||'')!==String(excludeRequestId||'')&&holdStillActive_(x.obj));
  if(!rows.length)return false;
  const catalog=getPublicConfig_().services;
  const durationFor=row=>{
    const ids=String(row['Service IDs']||'').split(',').filter(Boolean);
    const services=ids.map(id=>catalog.find(s=>s.id===id)).filter(Boolean);
    return services.length?services.reduce((sum,s)=>sum+Math.max(1,num_(s.duration)||30),0):30;
  };
  return rows.some(x=>{
    const candidates=[];
    if(normalizeDate_(x.obj['Appointment Date'])===date)candidates.push(x.obj['Appointment Time']);
    if(String(x.obj.Status)===CORE.STATUS.RESCHEDULE_REQUESTED&&normalizeDate_(x.obj['Requested Reschedule Date'])===date)candidates.push(x.obj['Requested Reschedule Time']);
    return candidates.some(existingTime=>{
      const existingStart=minutes_(existingTime);
      return existingStart>=0&&intervalsOverlap_(start,end,existingStart,existingStart+durationFor(x.obj));
    });
  });
}
function holdStillActive_(row) {
  const status=String(row.Status||'');
  const field=status===CORE.STATUS.REQUEST_PENDING ? 'Request Expires At' : status===CORE.STATUS.PAYMENT_PENDING ? 'Payment Expires At' : '';
  if(!field) return true;
  const raw=row[field];
  if(!raw) return true;
  const deadline=raw instanceof Date ? raw : new Date(raw);
  if(!Number.isFinite(deadline.getTime())) return true;
  return deadline.getTime()>Date.now();
}
function intervalsOverlap_(aStart,aEnd,bStart,bEnd) {
  return Number.isFinite(aStart)&&Number.isFinite(aEnd)&&Number.isFinite(bStart)&&Number.isFinite(bEnd)&&aStart<bEnd&&bStart<aEnd;
}
function minutes_(value) {
  const t=normalizeTime_(value).match(/^(\d{2}):(\d{2})$/);
  return t?Number(t[1])*60+Number(t[2]):-1;
}
function activeSlotSet_() {

/** Verify Telegram WebApp initData server-side; never trust client-supplied telegramId. */
function verifyTelegramInitData_(initData) {
  const raw=String(initData||'').trim();
  const token=prop_(CORE.BOT_TOKEN_PROP);
  if(!raw || !token) return null;
  try {
    const params={};
    raw.split('&').forEach(pair=>{
      const at=pair.indexOf('=');
      if(at<1)return;
      const key=decodeURIComponent(pair.slice(0,at).replace(/\+/g,' '));
      const value=decodeURIComponent(pair.slice(at+1).replace(/\+/g,' '));
      params[key]=value;
    });
    const suppliedHash=String(params.hash||'').toLowerCase();
    const authDate=Number(params.auth_date||0);
    const now=Math.floor(Date.now()/1000);
    if(!/^[a-f0-9]{64}$/.test(suppliedHash) || !authDate || authDate>now+60 || now-authDate>86400 || !params.user) return null;
    const checkString=Object.keys(params).filter(k=>k!=='hash').sort().map(k=>k+'='+params[k]).join('\n');
    const secretKey=Utilities.computeHmacSha256Signature(token,'WebAppData');
    const expectedHash=Utilities.computeHmacSha256Signature(checkString,secretKey).map(b=>(b<0?b+256:b).toString(16).padStart(2,'0')).join('');
    if(expectedHash!==suppliedHash) return null;
    const user=JSON.parse(params.user);
    if(!user || user.id===undefined || user.id===null || !/^\d+$/.test(String(user.id))) return null;
    return {id:String(user.id),username:String(user.username||'')};
  } catch(err) {
    console.error('Telegram initData verification failed',err);
    return null;
  }
}

  gy+=4*Math.floor(days/1461);days%=1461;

/**
 * Narrow, guarded migration for the two Telegram identity headers only.
 * Appends missing headers to CoreBookings; never rewrites existing headers or rows.
 * This function performs a write only after all preconditions pass.
 */
function addTelegramIdentityHeadersToStagingCoreBookings() {
  const expectedSheetName = 'KaenatChi Booking Core STAGING';
  const targetName = CORE.TABS.BOOKINGS;
  const newHeaders = ['Telegram Chat ID', 'Telegram Username'];
  const ss = staging_();

  if (ss.getName() !== expectedSheetName) {
    return {ok:false, changed:false, error:'WRONG_STAGING_SPREADSHEET',
      spreadsheet:ss.getName(), expectedSpreadsheet:expectedSheetName,
      message:'هیچ تغییری انجام نشد؛ نام فایل با محیط Staging مورد انتظار مطابقت ندارد.'};
  }

  const sh = ss.getSheetByName(targetName);
  if (!sh) {
    return {ok:false, changed:false, error:'COREBOOKINGS_MISSING',
      message:'هیچ تغییری انجام نشد؛ شیت CoreBookings وجود ندارد.'};
  }

  const expected = CORE.HEADERS.CoreBookings.slice();
  const legacyHeaders = expected.filter(h => newHeaders.indexOf(h) < 0);
  const lastColumn = sh.getLastColumn();
  if (!lastColumn || !sh.getLastRow()) {
    return {ok:false, changed:false, error:'COREBOOKINGS_EMPTY',
      message:'هیچ تغییری انجام نشد؛ شیت CoreBookings خالی است و نیاز به بررسی جداگانه دارد.'};
  }

  const current = sh.getRange(1, 1, 1, lastColumn)
    .getDisplayValues()[0].map(v => String(v || '').trim());

  // Require the existing legacy header sequence to match exactly. Permit only
  // either or both target headers appended at the end by a prior partial run.
  const prefixMatches = legacyHeaders.every((h, i) => current[i] === h);
  const trailing = current.slice(legacyHeaders.length).filter(Boolean);
  const noUnexpectedNonblank = trailing.every(h => newHeaders.indexOf(h) >= 0);
  const noDuplicates = new Set(current.filter(Boolean)).size === current.filter(Boolean).length;
  const targetHeadersAlreadyPresent = newHeaders.filter(h => current.indexOf(h) >= 0);
  const targetHeadersAreTrailing = targetHeadersAlreadyPresent.every(h =>
    current.indexOf(h) >= legacyHeaders.length
  );

  if (!prefixMatches || !noUnexpectedNonblank || !noDuplicates || !targetHeadersAreTrailing) {
    return {ok:false, changed:false, error:'COREBOOKINGS_HEADER_GUARD_FAILED',
      currentHeaders:current,
      message:'هیچ تغییری انجام نشد؛ هدرهای فعلی با الگوی امن مورد انتظار تطبیق ندارند.'};
  }

  const missing = newHeaders.filter(h => current.indexOf(h) < 0);
  if (!missing.length) {
    return {ok:true, changed:false, spreadsheet:ss.getName(), sheet:targetName,
      headersAdded:[], message:'هر دو هدر از قبل موجودند؛ تغییری لازم نیست.'};
  }

  // Single write: append only the missing header cells. Existing rows/columns
  // and all other tabs remain untouched.
  sh.getRange(1, lastColumn + 1, 1, missing.length).setValues([missing]);

  return {ok:true, changed:true, spreadsheet:ss.getName(), sheet:targetName,
    headersAdded:missing, previousLastColumn:lastColumn,
    newLastColumn:lastColumn + missing.length,
    message:'فقط هدرهای شناسایی تلگرام به انتهای ردیف اول CoreBookings اضافه شدند.'};
}

/**
 * Read-only preflight for an existing STAGING spreadsheet.
 * Does not create sheets, write headers, mutate Script Properties, call Telegram,
 * or contact CMS/VIP. Use this before any schema setup or migration.
 */
function auditStagingSchema() {
  const required = Object.keys(CORE.HEADERS).reduce((out, name) => {
    out[name] = CORE.HEADERS[name].slice();
    return out;
  }, {});
  required[CORE.TABS.CLOSURES] = ['closure_id','start_datetime','end_datetime','reason','source','active','created_at','updated_at'];
  const report = {
    ok: false,
    readOnly: true,
    spreadsheetName: '',
    properties: {},
    sheets: {},
    blockers: []
  };
  ['STAGING_SPREADSHEET_ID','ADMIN_KEY','CMS_API_URL','VIP_SPREADSHEET_ID',
   'KAENATCHI_TELEGRAM_BOT_TOKEN','KAENATCHI_TELEGRAM_WEBHOOK_SECRET',
   'KAENATCHI_TELEGRAM_ADMIN_CHAT_ID'].forEach(key => {
    report.properties[key] = Boolean(prop_(key));
  });
  let ss;
  try {
    const id = prop_(CORE.STAGING_PROP);
    if (!id) fail_('STAGING_ID_MISSING','STAGING_SPREADSHEET_ID is not configured.');
    ss = SpreadsheetApp.openById(id);
    report.spreadsheetName = ss.getName();
  } catch (err) {
    report.blockers.push('STAGING_SPREADSHEET_UNAVAILABLE');
    report.ok = false;
    return report;
  }
  Object.keys(required).forEach(name => {
    const sh = ss.getSheetByName(name);
    if (!sh) {
      report.sheets[name] = {exists:false, missingHeaders:required[name]};
      report.blockers.push('MISSING_SHEET:' + name);
      return;
    }
    const width = Math.max(1, sh.getLastColumn());
    const headers = sh.getLastRow()
      ? sh.getRange(1,1,1,width).getDisplayValues()[0].map(v => String(v || '').trim()).filter(Boolean)
      : [];
    const missing = required[name].filter(h => headers.indexOf(h) < 0);
    report.sheets[name] = {
      exists:true,
      lastRow:sh.getLastRow(),
      headers,
      missingHeaders:missing
    };
    if (missing.length) report.blockers.push('SCHEMA_MISMATCH:' + name);
  });
  if (!report.properties.CMS_API_URL) report.blockers.push('MISSING_PROPERTY:CMS_API_URL');
  if (!report.properties.ADMIN_KEY) report.blockers.push('MISSING_PROPERTY:ADMIN_KEY');
  report.ok = report.blockers.length === 0;
  console.log(JSON.stringify(report, null, 2));
  return report;
}


/**
 * Explicit, isolated STAGING schema initializer.
 * It never alters existing legacy tabs. It validates all existing target tabs
 * before creating any missing Core* tabs, then writes headers only to newly
 * created tabs. Do not run until auditStagingSchema() has been reviewed.
 */
function initializeIsolatedStagingSchema() {
  const ss = staging_();
  const expected = {
    CoreBookings: CORE.HEADERS.CoreBookings,
    CorePayments: CORE.HEADERS.CorePayments,
    CoreBookingLogs: CORE.HEADERS.CoreBookingLogs,
    Closures: ['closure_id','start_datetime','end_datetime','reason','source','active','created_at','updated_at']
  };
  const plan = [];
  const blockers = [];
  Object.keys(expected).forEach(name => {
    const sh = ss.getSheetByName(name);
    if (!sh) {
      plan.push({name, action:'create'});
      return;
    }
    const headers = sh.getLastRow()
      ? sh.getRange(1,1,1,Math.max(1,sh.getLastColumn())).getDisplayValues()[0].map(v => String(v||'').trim()).filter(Boolean)
      : [];
    if (!headers.length) {
      plan.push({name, action:'initialize-empty-existing'});
      return;
    }
    const missing = expected[name].filter(h => headers.indexOf(h) < 0);
    if (missing.length) blockers.push({name, missing});
    else plan.push({name, action:'keep-compatible-existing'});
  });
  if (blockers.length) {
    return {ok:false, changed:false, error:'SCHEMA_MISMATCH', blockers,
      message:'هیچ تغییری انجام نشد؛ ابتدا ناسازگاری شیت‌های Core موجود را بررسی کن.'};
  }
  // All target schemas were checked before the first write.
  plan.filter(x => x.action==='create').forEach(x => {
    const sh = ss.insertSheet(x.name);
    sh.getRange(1,1,1,expected[x.name].length).setValues([expected[x.name]]);
  });
  plan.filter(x => x.action==='initialize-empty-existing').forEach(x => {
    const sh = ss.getSheetByName(x.name);
    sh.getRange(1,1,1,expected[x.name].length).setValues([expected[x.name]]);
  });
  console.log(JSON.stringify({ok:true, changed:plan.some(x=>x.action==='create'||x.action==='initialize-empty-existing'),
  spreadsheet:ss.getName(), plan, legacyTabsPreserved:true}, null, 2));
return {ok:true, changed:plan.some(x=>x.action==='create'||x.action==='initialize-empty-existing'),
    spreadsheet:ss.getName(), plan, legacyTabsPreserved:true,
    message:'فقط شیت‌های Core خالی یا تازه ایجاد شدند؛ شیت‌های قدیمی دست‌نخورده ماندند.'};
}

    const config = getPublicConfig_();

    report.ok = config.ok === true;
    report.serviceCount = config.services.length;

    report.services = config.services.map(function (service) {
      return {
        id: service.id,
        name: service.name,
        category: service.category,
        price: service.price,
        duration: service.duration
      };
    });

    report.availableDates = config.availableDates;
    report.bookingRules = config.bookingRules;

  } catch (err) {
    report.ok = false;
    report.error = String(
      err && err.message ? err.message : err
    ).slice(0, 300);
  }

  console.log(JSON.stringify(report, null, 2));
  return report;
}

function diagnoseStagingCmsRawServices() {
  const report = {
    executed: true,
    readOnly: true,
    endpoint: 'getMiniAppData'
  };

  try {
    const api = prop_(CORE.CMS_API_URL_PROP);

    if (!api) {
      throw new Error('CMS_API_URL در تنظیمات Staging موجود نیست.');
    }

    const separator = api.indexOf('?') >= 0 ? '&' : '?';
    const response = UrlFetchApp.fetch(
      api + separator + 'action=getMiniAppData',
      {
        muteHttpExceptions: true,
        followRedirects: true
      }
    );

    report.httpStatus = response.getResponseCode();

    if (
      report.httpStatus < 200 ||
      report.httpStatus >= 300
    ) {
      report.bodyPreview = response
        .getContentText()
        .slice(0, 1000);

      throw new Error('پاسخ HTTP موفق نبود.');
    }

    const payload = JSON.parse(response.getContentText());

    report.success = payload.success === true;
    report.topLevelKeys = Object.keys(payload);
    report.servicesIsArray = Array.isArray(payload.services);
    report.serviceCount = Array.isArray(payload.services)
      ? payload.services.length
      : null;

    if (Array.isArray(payload.services)) {
      report.samples = payload.services
        .slice(0, 10)
        .map(function (service, index) {
          return {
            index: index,
            keys: Object.keys(service),
            data: service
          };
        });
    }

    report.ok = report.success && report.servicesIsArray;

  } catch (err) {
    report.ok = false;
    report.error = String(
      err && err.message ? err.message : err
    ).slice(0, 500);
  }

  console.log(JSON.stringify(report, null, 2));
  return report;
}

function diagnoseStagingCmsFilter() {
  const report = {
    executed: true,
    readOnly: true,
    rows: []
  };

  try {
    const api = prop_(CORE.CMS_API_URL_PROP);
    if (!api) {
      throw new Error('CMS_API_URL تنظیم نشده است.');
    }

    const separator = api.indexOf('?') >= 0 ? '&' : '?';
    const response = UrlFetchApp.fetch(
      api + separator + 'action=getMiniAppData',
      {
        muteHttpExceptions: true,
        followRedirects: true
      }
    );

    report.httpStatus = response.getResponseCode();

    if (report.httpStatus < 200 || report.httpStatus >= 300) {
      throw new Error('پاسخ HTTP ناموفق است.');
    }

    const payload = JSON.parse(response.getContentText());

    if (!payload || !Array.isArray(payload.services)) {
      throw new Error('آرایه services در پاسخ وجود ندارد.');
    }

    report.serviceCount = payload.services.length;

    report.rows = payload.services.map(function (r, index) {
      const name = String(
        first_(r, [
          'نام خدمت', 'عنوان', 'نام',
          'Service Name', 'Name', 'Title', 'title'
        ]) || ''
      ).trim();

      const explicitId = String(
        first_(r, [
          'شناسه', 'شناسه خدمت', 'Service ID',
          'ServiceId', 'ID', 'Id', 'id', 'slug'
        ]) || ''
      ).trim();

      const rawPrice = first_(r, [
        'قیمت', 'هزینه', 'Price',
        'Base Price', 'Original Price', 'مبلغ', 'price'
      ]);

      const active = String(
        first_(r, [
          'فعال', 'وضعیت', 'Active',
          'active', 'Status', 'status'
        ]) || ''
      ).trim().toLocaleLowerCase('fa');

      const normalizedName = name.replace(
        /[\u200c\u200f\u200e\s_-]+/g,
        ''
      ).toLocaleLowerCase('fa');

      const id = explicitId || ('diagnostic-row-' + (index + 1));

      const hasPrice = rawPrice !== undefined &&
        rawPrice !== null &&
        String(rawPrice).trim() !== '';

      const isActive = !active ||
        ['بله', 'فعال', 'true', 'yes', '1', 'active']
          .includes(active);

      const aggregate = [
        'انرژیخوانی', 'شمعتراپی', 'سایکوتراپی'
      ].includes(normalizedName);

      const reasons = [];

      if (!id) reasons.push('شناسه خالی');
      if (!name) reasons.push('نام خالی');
      if (!isActive) reasons.push('وضعیت فعال پذیرفته نشد');
      if (!hasPrice) reasons.push('قیمت خالی');
      if (aggregate) reasons.push('ردیف دسته‌بندی کلی');

      return {
        index: index,
        name: name,
        id: id,
        activeValue: active,
        rawPrice: rawPrice,
        hasPrice: hasPrice,
        isActive: isActive,
        aggregate: aggregate,
        acceptedByExpectedFilter:
          Boolean(id && name && isActive && hasPrice && !aggregate),
        rejectionReasons: reasons
      };
    });

    report.acceptedCount = report.rows.filter(
      function (r) {
        return r.acceptedByExpectedFilter;
      }
    ).length;

    report.ok = true;

  } catch (err) {
    report.ok = false;
    report.error = String(
      err && err.message ? err.message : err
    ).slice(0, 500);
  }

  console.log(JSON.stringify(report, null, 2));
  return report;
}

function bookingSlug_(value) {
  return String(value || '')
    .trim()
    .toLowerCase()
    .replace(/[آأإ]/g, 'ا')
    .replace(/[^a-z0-9\u0600-\u06ff]+/g, '-')
    .replace(/^-+|-+$/g, '') || 'item';
}
